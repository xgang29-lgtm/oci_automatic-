DB_FILE = "oci_manager.db"


def init_db():
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()

    cursor.execute('''
        CREATE TABLE IF NOT EXISTS oci_accounts (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            account_name TEXT UNIQUE,
            region TEXT,
            tenancy_ocid TEXT,
            user_ocid TEXT,
            fingerprint TEXT,
            private_key_path TEXT,
            assigned_username TEXT DEFAULT 'admin',
            customer TEXT DEFAULT 'Yellowstone',
            password TEXT DEFAULT '123456',
            role TEXT DEFAULT 'normal',
            api_key TEXT DEFAULT ''
        )
    ''')

    # 兼容性安全检查：确保旧数据库表结构自动补齐所有新增字段
    cursor.execute("PRAGMA table_info(oci_accounts)")
    columns = [col[1] for col in cursor.fetchall()]
    if "customer" not in columns:
        cursor.execute("ALTER TABLE oci_accounts ADD COLUMN customer TEXT DEFAULT 'Yellowstone'")
    if "password" not in columns:
        cursor.execute("ALTER TABLE oci_accounts ADD COLUMN password TEXT DEFAULT '123456'")
    if "role" not in columns:
        cursor.execute("ALTER TABLE oci_accounts ADD COLUMN role TEXT DEFAULT 'normal'")
    if "api_key" not in columns:
        cursor.execute("ALTER TABLE oci_accounts ADD COLUMN api_key TEXT DEFAULT ''")

    cursor.execute('''
        CREATE TABLE IF NOT EXISTS customers (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            customer_name TEXT UNIQUE,
            tenancy_ocid TEXT,
            tenancy_name TEXT
        )
    ''')

    cursor.execute("SELECT COUNT(*) FROM customers")
    if cursor.fetchone()[0] == 0:
        default_customers = [
            ("Yellowstone", "ocid1.tenancy.oc1..yellowstone001", "yellowstone-prod-tenancy"),
            ("Apollo", "ocid1.tenancy.oc1..apollo002", "apollo-ai-tenancy"),

            ("telesisb", "ocid1.tenancy.oc1..telesisb004", "telesisb-edge-tenancy")

        CREATE TABLE IF NOT EXISTS system_users (

security = HTTPBasic()


def get_current_user(credentials: HTTPBasicCredentials = Depends(security)):
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    cursor.execute(
        "SELECT username, email, role, permission FROM system_users WHERE username = ? AND password = ?",
        (credentials.username, credentials.password)
    )
    user = cursor.fetchone()
    conn.close()

    if not user:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="用户名或密码错误，请检查认证凭据",
            headers={"WWW-Authenticate": "Basic"},
        )
    return {
        "username": user[0],
        "email": user[1],
        "role": user[2],
        "permission": user[3]
    }


def get_all_customer_names() -> List[str]:
    """实时从 customers 表读取所有已加入的客户名称列表，实现全局实时同步"""
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    cursor.execute("SELECT customer_name FROM customers ORDER BY id ASC")
    rows = cursor.fetchall()
    conn.close()
    return [r[0] for r in rows] if rows else ["Yellowstone", "Apollo", "telesis", "telesisb"]

@app.post("/api/rag/query")
async def api_delivery_rag_query(
    query: str = Form(...),
    sources: str = Form("internet"),
    files: List[UploadFile] = File(default=[]),
    current_user: dict = Depends(get_current_user)
):
    selected_sources = [s.strip() for s in sources.split(",") if s.strip()]
    retrieved_contexts = []
    ingested_files = []

    # 1. 处理用户通过页面 "+ 上传本地文件" 传入的文件
    for upload_file in files:
        if upload_file.filename:
            raw_bytes = await upload_file.read()
            parsed_text = await asyncio.to_thread(parse_uploaded_file, raw_bytes, upload_file.filename)
            await asyncio.to_thread(ingest_to_chromadb, upload_file.filename, parsed_text, "本地上传文件")
            ingested_files.append(upload_file.filename)

    # 2. 从本地 ChromaDB 向量库中检索
    if "upload" in selected_sources or ingested_files:
        local_hits = await asyncio.to_thread(query_local_chromadb, query, 4)
        retrieved_contexts.extend(local_hits)

    # 3. 实时检索 OCI 内部 Confluence
    if "confluence" in selected_sources:
        cf_hits = await asyncio.to_thread(query_oci_confluence, query, 3)
        retrieved_contexts.extend(cf_hits)

    # 4. 调用 Gemini 模型（若勾选 "internet"，自动启用联网 Grounding）
    use_internet = "internet" in selected_sources
    gemini_res = await asyncio.to_thread(
        generate_gemini_rag_response,
        query=query,
        contexts=retrieved_contexts,
        use_internet=use_internet
    )

    references = [
        {"source": c["source"], "title": c["title"], "url": c["url"]}
        for c in retrieved_contexts
    ] + gemini_res["web_refs"]

    return {
        "status": "ok",
        "query": query,
        "ingested_files": ingested_files,
        "answer": gemini_res["answer"],
        "references": references
    }

@app.post("/users/update_settings")
def update_settings(
    new_password: Optional[str] = Form(None),
    new_email: Optional[str] = Form(None),
    current_user: dict = Depends(get_current_user)
):
    conn = sqlite3.connect(DB_FILE)
    cursor = conn.cursor()
    if new_password and new_email:
        cursor.execute("UPDATE system_users SET password = ?, email = ? WHERE username = ?", (new_password, new_email, current_user["username"]))
    elif new_password:
        cursor.execute("UPDATE system_users SET password = ? WHERE username = ?", (new_password, current_user["username"]))
    elif new_email:
        cursor.execute("UPDATE system_users SET email = ? WHERE username = ?", (new_email, current_user["username"]))
    conn.commit()
    conn.close()
    return RedirectResponse(url="/", status_code=303)

@app.post("/api/oci/resources")
def get_tenancy_resources_and_audit(
    tenancy_ocid: str = Form(...)
):
    try:
        # 1. 加载本地 OCI 配置文件（确保 ~/.oci/config 存在且有对应权限）
        config = oci.config.from_file()
        
        # 如果前端传入了特定的 Tenancy OCID，可以在此处进行校验或动态适配
        # 注意：若需跨 Tenancy 查询，需确保当前 API Key 具备对应租户的 inspect 权限
        
        # 2. 初始化 OCI 资源搜索客户端 (Resource Search)
        search_client = oci.resource_search.ResourceSearchClient(config)
        
        # 构造结构化查询语句，实时拉取该 Tenancy 下的所有云资源
        search_query = oci.resource_search.models.StructuredSearchDetails(
            type="Structured",
            query="query all resources"
        )
        
        search_response = search_client.search_resources(search_query)
        raw_items = search_response.data.items if search_response.data else []

        resources = []
        for item in raw_items:
            resources.append({
                "display_name": getattr(item, "display_name", "Unnamed Resource"),
                "resource_type": getattr(item, "resource_type", "Unknown"),
                "lifecycle_state": getattr(item, "lifecycle_state", "UNKNOWN"),
                "compartment_id": getattr(item, "compartment_id", "N/A"),
                "identifier": getattr(item, "identifier", "N/A"),
                "time_created": str(getattr(item, "time_created", "N/A"))
            })

        # 3. 初始化 OCI 审计客户端 (Audit Service) 获取工作流审计日志
        # 审计服务需要指定 Tenancy OCID
        audit_client = oci.audit.AuditClient(config)
        
        end_time = datetime.utcnow()
        start_time = end_time - timedelta(days=7) # 获取最近 7 天的审计日志
        
        audit_response = audit_client.list_events(
            compartment_id=tenancy_ocid, # 传入目标 Tenancy/Compartment OCID
            start_time=start_time,
            end_time=end_time
        )
        
        audit_events = []
        for event in (audit_response.data[:20] if audit_response.data else []): # 取最近 20 条
            audit_events.append({
                "event_name": getattr(event, "event_name", "Action"),
                "actor": getattr(event, "identity", {}).get("principal_name", "System"),
                "time": str(getattr(event, "event_time", "")),
                "status": "SUCCESS" # OCI 审计事件默认记录执行动作
            })

        return {
            "status": "success",
            "resources": resources,
            "audit_logs": audit_events
        }

    except Exception as e:
        # 打印详细错误到终端方便排查
        print(f"[OCI Sync Error] {str(e)}")
        raise HTTPException(status_code=500, detail=f"同步 OCI 实时数据失败: {str(e)}")
